# Constraints file for installing whisnemo[diarize] on Windows.
#
# Pip's resolver, when installing nemo-toolkit and its transitive
# dependencies, will otherwise upgrade torch and numpy to versions that
# break other pinned packages in the diarize stack. Pinning them here
# forces pip to find a compatible install plan around these versions.
#
# Versions match the validated production environment on Linux.
#
# Usage (Windows):
#   pip install torch==2.11.0 torchaudio==2.11.0
#   pip install -c constraints/windows.txt "whisnemo[diarize] @ git+https://github.com/humanlab/WhisNemo.git"

torch==2.11.0
torchaudio==2.11.0
numpy==1.23.5
